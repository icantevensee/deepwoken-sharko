import os
import tkinter as tk
import psutil

class MainClass:
    def __init__(self):
        self.root = tk.Tk()
        self.root.title("Leakymenu")
        
        self.width = 357
        self.height = 342
        self.root.geometry(f"{self.width}x{self.height}")
        self.root.resizable(False, False)

        self.root.overrideredirect(True)
        self.root.attributes("-topmost", True)
        self.root.config(bg="#000001")
        self.root.attributes("-transparentcolor", "#000001")
        
        self._create_gui()

        self.process = psutil.Process(os.getpid())
        self.log_stats()

        self.root.mainloop()

    def log_stats(self):
        count = len(self.root.winfo_children())
        mem_mb = self.process.memory_info().rss / (1024 * 1024)
        print(f"RAM: {mem_mb:.2f} MB | Tkinter Objects: {count}")
        self.root.after(500, self.log_stats)

    def _create_gui(self):
        self.canvas = tk.Canvas(
            self.root, 
            width=self.width, 
            height=self.height, 
            bg="blue", 
            highlightthickness=0
        )
        self.canvas.pack(fill=tk.BOTH, expand=True)

        self.canvas.bind("<Double-Button-2>", self.show_popup_menu)

        self.menu = tk.Menu(self.root, tearoff=0)

        self.movie_menu = tk.Menu(
            self.menu, 
            tearoff=0,
            bg="#2a2a2a",
            fg="#ffffff",
            activebackground="#0078d4",
            activeforeground="#ffffff",
            bd=1
        )
        
        self.movie_menu.add_command(label="option1", command=self.option1)
        self.movie_menu.add_command(label="option2", command=self.option2)
        self.movie_menu.add_command(label="option3", command=self.option3)
        
        self.menu.add_cascade(label="menu1", menu=self.movie_menu)

    def show_popup_menu(self, event):
        try:
            self.menu.tk_popup(event.x_root, event.y_root)
        finally:
            self.menu.grab_release()

    def option1(self):
        pass

    def option2(self):
        pass

    def option3(self):
        pass

if __name__ == "__main__":
    main = MainClass()
